# Cross build for FB_BOARD=stm32g4: bootloader + per-node, per-slot application images for STM32G431RB.
# UNVERIFIED ON HARDWARE: this configuration compiles and links; nothing here has run on a board.
add_link_options(--specs=nano.specs --specs=nosys.specs -Wl,--gc-sections -Wl,--print-memory-usage)
set(FB_VERSION 1 CACHE STRING "Firmware image version stamped into signed images")
option(FB_BENCH_FAULT_IMAGES "Also build deliberately broken images (hang / never healthy). BENCH ONLY." OFF)

set(HAL ${TP}/stm32g4xx_hal_driver)
set(CMSIS_DEV ${TP}/cmsis_device_g4)
set(CMSIS_CORE ${TP}/cmsis_core/CMSIS/Core/Include)
set(HAL_MODS "" hal_cortex hal_rcc hal_rcc_ex hal_gpio hal_pwr hal_pwr_ex hal_flash hal_flash_ex hal_fdcan
    hal_i2c hal_i2c_ex hal_tim hal_tim_ex hal_adc hal_adc_ex hal_dma hal_dma_ex hal_exti)
set(HAL_SRC ${HAL}/Src/stm32g4xx_hal.c)
foreach(m ${HAL_MODS})
  if(m)
    list(APPEND HAL_SRC ${HAL}/Src/stm32g4xx_${m}.c)
  endif()
endforeach()
add_library(hal STATIC ${HAL_SRC} ${CMSIS_DEV}/Source/Templates/system_stm32g4xx.c)
target_compile_definitions(hal PUBLIC STM32G431xx USE_HAL_DRIVER)
target_include_directories(hal PUBLIC ${HAL}/Inc ${CMSIS_DEV}/Include ${CMSIS_CORE} ${CMAKE_CURRENT_SOURCE_DIR}/../../bsp ${CMAKE_CURRENT_SOURCE_DIR}/../../bsp/stm32g4)
target_compile_options(hal PRIVATE -w)

set(RTOS ${TP}/freertos)
add_library(freertos STATIC ${RTOS}/tasks.c ${RTOS}/queue.c ${RTOS}/list.c ${RTOS}/portable/GCC/ARM_CM4F/port.c ${RTOS}/portable/MemMang/heap_4.c)
target_include_directories(freertos PUBLIC ${RTOS}/include ${RTOS}/portable/GCC/ARM_CM4F ${CMAKE_CURRENT_SOURCE_DIR}/../../bsp ${CMAKE_CURRENT_SOURCE_DIR}/../../bsp/stm32g4)
target_compile_options(freertos PRIVATE -w)

set(STARTUP ${CMSIS_DEV}/Source/Templates/gcc/startup_stm32g431xx.s)
# bsp_boot: what the bootloader may use (clock, LED/GPIO, IWDG, flash). bsp: the full set for applications.
add_library(bsp_boot STATIC ../../bsp/stm32g4/bsp_clock.c ../../bsp/stm32g4/bsp_flash.c)
target_link_libraries(bsp_boot PUBLIC hal fb_common)
target_include_directories(bsp_boot PUBLIC ../../bsp ../../bsp/stm32g4)
target_compile_options(bsp_boot PRIVATE -Wall -Wextra)
add_library(bsp STATIC ../../bsp/stm32g4/bsp_core.c ../../bsp/stm32g4/bsp_can.c ../../bsp/stm32g4/bsp_sensors.c)
target_link_libraries(bsp PUBLIC bsp_boot)
target_compile_options(bsp PRIVATE -Wall -Wextra)

function(fb_finish tgt)
  add_custom_command(TARGET ${tgt} POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:${tgt}> $<TARGET_FILE:${tgt}>.bin
    COMMAND ${CMAKE_SIZE} $<TARGET_FILE:${tgt}>
    COMMENT "${tgt}: .bin + size")
endfunction()

# ---- bootloader (one per node id because it only boots images built for its node)
foreach(role battery body telematics)
  if(role STREQUAL battery)
    set(NID 1)
  elseif(role STREQUAL body)
    set(NID 2)
  else()
    set(NID 3)
  endif()
  add_executable(bootloader_${role} ../../bootloader/boot_main.c ../../bsp/stm32g4/bsp_boot.c ${STARTUP})
  target_compile_definitions(bootloader_${role} PRIVATE BOOT_NODE_ID=${NID} USER_VECT_TAB_ADDRESS VECT_TAB_OFFSET=0x0)
  target_link_libraries(bootloader_${role} bsp_boot)
  target_link_options(bootloader_${role} PRIVATE -T${CMAKE_CURRENT_SOURCE_DIR}/boot.ld -Wl,-Map=bootloader_${role}.map)
  set_target_properties(bootloader_${role} PROPERTIES SUFFIX ".elf" LINK_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/boot.ld)
  fb_finish(bootloader_${role})
endforeach()

# ---- applications: linked once per slot; sign_image.py then prepends the 0x200-byte signed header
function(add_fb_app role slot variant)
  if(slot EQUAL 0)
    set(letter A)
    set(ORIGIN 0x08008200)
    set(VOFF 0x8200)
  else()
    set(letter B)
    set(ORIGIN 0x08014200)
    set(VOFF 0x14200)
  endif()
  set(LEN "48640")   # 48 KB slot - 0x200 header
  set(name app_${role}_${letter}${variant})
  set(APP_ORIGIN ${ORIGIN})
  set(APP_LENGTH ${LEN})
  configure_file(${CMAKE_CURRENT_SOURCE_DIR}/app.ld.in ${CMAKE_CURRENT_BINARY_DIR}/${name}.ld @ONLY)
  string(TOUPPER ${role} ROLE_UP)
  if(role STREQUAL telematics)
    add_executable(${name} ../../app/telematics/telematics_main.c ../../app/telematics/car_tx.c ${STARTUP})
  else()
    add_executable(${name} ../../app/app_main.c ../../app/${role}/${role}_hw.c ${STARTUP})
  endif()
  target_include_directories(${name} PRIVATE ../../app ../../keys)
  target_compile_definitions(${name} PRIVATE FB_ROLE_${ROLE_UP} APP_SLOT=${slot} USER_VECT_TAB_ADDRESS VECT_TAB_OFFSET=${VOFF})
  if(variant STREQUAL "_hang")
    target_compile_definitions(${name} PRIVATE FB_BENCH_HANG_AT_BOOT)
  elseif(variant STREQUAL "_unhealthy")
    target_compile_definitions(${name} PRIVATE FB_BENCH_NEVER_HEALTHY)
  endif()
  target_link_libraries(${name} bsp freertos)
  target_link_options(${name} PRIVATE -T${CMAKE_CURRENT_BINARY_DIR}/${name}.ld -Wl,-Map=${name}.map)
  set_target_properties(${name} PROPERTIES SUFFIX ".elf" LINK_DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/${name}.ld)
  fb_finish(${name})
endfunction()

foreach(role battery body telematics)
  foreach(slot 0 1)
    add_fb_app(${role} ${slot} "")
    if(FB_BENCH_FAULT_IMAGES AND NOT role STREQUAL telematics)
      add_fb_app(${role} ${slot} "_hang")
      add_fb_app(${role} ${slot} "_unhealthy")
    endif()
  endforeach()
endforeach()
